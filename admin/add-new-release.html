import { db } from './firebase-config.js'; import { collection, addDoc, serverTimestamp, getDocs, deleteDoc, doc, query, orderBy } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js"; const form = document.getElementById('new-release-form'); const newReleaseList = document.getElementById('new-release-list'); // 1. Fetch & Display Existing New Releases async function loadNewReleases() { if (!newReleaseList) return; newReleaseList.innerHTML = '

Loading current New Releases…

'; try { const q = query(collection(db, "movies"), orderBy("createdAt", "desc")); const snapshot = await getDocs(q); if (snapshot.empty) { newReleaseList.innerHTML = '

No New Releases added yet. Add your first one below.

'; return; } let html = ''; snapshot.forEach((docSnap) => { const data = docSnap.data(); html += `
${data.title}
Msg ID: ${data.telegramMsgId || data.messageId || 'N/A'} | ${data.year || '2026'}
`; }); newReleaseList.innerHTML = html; // Attach Delete Listeners document.querySelectorAll('.btn-delete').forEach(btn => { btn.addEventListener('click', async (e) => { const id = e.target.getAttribute('data-id'); if (confirm("Are you sure you want to delete this release?")) { await deleteDoc(doc(db, "movies", id)); loadNewReleases(); } }); }); } catch (err) { console.error("Error fetching new releases:", err); newReleaseList.innerHTML = '

Error loading releases.

'; } } // 2. Safe Form Submit Event if (form) { form.addEventListener('submit', async (e) => { e.preventDefault(); // Safe getters with fallbacks (no null errors) const title = document.getElementById('nr-title')?.value?.trim() || ''; const posterUrl = document.getElementById('nr-poster-url')?.value?.trim() || ''; const telegramMsgId = document.getElementById('nr-telegram-msg-id')?.value?.trim() || ''; const year = document.getElementById('nr-year')?.value?.trim() || '2026'; const format = document.getElementById('nr-format')?.value?.trim() || '1080p HD'; const language = document.getElementById('nr-language')?.value?.trim() || 'Hindi'; const summary = document.getElementById('nr-summary')?.value?.trim() || ''; if (!title || !posterUrl || !telegramMsgId) { alert("Please fill in Title, Poster URL, and Telegram Link/Message ID."); return; } const submitBtn = form.querySelector('button[type="submit"]'); if (submitBtn) submitBtn.disabled = true; try { await addDoc(collection(db, "movies"), { title: title, posterUrl: posterUrl, telegramMsgId: telegramMsgId, messageId: telegramMsgId, year: year, quality: format, language: language, summary: summary, category: "New Releases", isNewRelease: true, createdAt: serverTimestamp() }); alert("New Release added successfully!"); form.reset(); loadNewReleases(); } catch (err) { console.error("Error adding movie:", err); alert("Failed to add movie: " + err.message); } finally { if (submitBtn) submitBtn.disabled = false; } }); } // Initial Call loadNewReleases();
